# Shanaya Vercel Environment Variables
# =====================================
# Paste these into Vercel: Settings → Environment Variables
# Assign to: server service (Production environment)
# Then redeploy.

# === REQUIRED (server will crash without these) ===
SHANAYA_API_TOKEN=<paste output of: openssl rand -hex 32>
NODE_ENV=production

# === RECOMMENDED (for cloud inference to work) ===
GROQ_API_KEY=<your Groq API key from console.groq.com>
CLIENT_ORIGIN=https://shanaya.eliteforums.in

# === OPTIONAL (enhanced features) ===
GROQ_MODEL=openai/gpt-oss-120b
GROQ_MODEL_FALLBACKS=openai/gpt-oss-20b,llama-3.1-8b-instant
VITE_API_BASE_URL=https://shanaya.eliteforums.in

# === HOW TO GENERATE THE TOKEN ===
# Run this in your terminal:
#   openssl rand -hex 32
# Copy the output and paste it as the value for SHANAYA_API_TOKEN
